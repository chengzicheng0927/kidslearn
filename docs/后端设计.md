## 2. 后端开发规范

> 继承自 `CLAUDE.md` 中的 JDK 21 现代化语法规约。

### 2.1 虚拟线程（Virtual Threads）

```yaml
# application.yml
spring:
  threads:
    virtual:
      enabled: true
```

- **禁止** 手动创建 `FixedThreadPool`，统一使用虚拟线程 scoped values
- **禁止** 在虚拟线程中长时间持有 `synchronized` 锁，优先使用 `ReentrantLock`
- 推荐使用 `ExecutorService` 的虚拟线程实现：`Executors.newVirtualThreadPerTaskExecutor()`

### 2.2 模式匹配（Pattern Matching）

```java
// 禁止显式强转，必须使用 instanceof + record pattern
@Override
public ProblemDetail handleMethodArgumentNotValid(MethodArgumentNotValidException ex) {
    String detail = switch (ex.getErrorCount()) {
        case 0 -> "无效请求";
        case 1 -> ex.getFieldErrors().getFirst().getDefaultMessage();
        default -> "%d 个字段校验失败".formatted(ex.getErrorCount());
    };
    // ...
}

// record 作为 DTO/VO/Event
public record QuizSubmitRequest(
    @NotBlank String questionId,
    String planQuestionId,        // 关联的计划题目ID（可选，用于学习计划场景）
    @NotBlank String answerId,
    @Min(0) int timeSpent
) {}

public record QuizSubmitResponse(
    boolean isCorrect,
    String resultMode,
    String correctAnswerId,
    String explanation,
    List<String> explanationSteps,
    String encouragement,
    boolean showResult,
    boolean hasMore,
    Result result
) {}
```

### 2.5 异常处理（RFC 7807 ProblemDetail）

```yaml
# application.yml
spring:
  mvc:
    problemdetails:
      enabled: true
```

```java
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ProblemDetail handleValidation(MethodArgumentNotValidException ex) {
        var field = ex.getFieldErrors().getFirst();
        return ProblemDetail.forStatusAndDetail(
            HttpStatus.BAD_REQUEST,
            field.getDefaultMessage()
        );
    }

    @ExceptionHandler(PlanNotFoundException.class)
    public ProblemDetail handleNotFound(PlanNotFoundException ex) {
        return ProblemDetail.forStatusAndDetail(
            HttpStatus.NOT_FOUND,
            ex.getMessage()
        );
    }
}
```

### 2.6 可观测性

```java
public class QuizService {
    private final ObservationRegistry registry;

    public QuizSubmitResponse submit(QuizSubmitRequest req) {
        return Observation.start("quiz.submit", registry)
            .contextualName("submit-answer")
            .observe(() -> doSubmit(req));
    }
}
```

```sql
-- 题目表（由 LLM 动态生成，存储供缓存和去重）
CREATE TABLE questions (
    id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    -- 题目基本信息
    subject             VARCHAR(5) NOT NULL CHECK (subject IN ('en', 'ma')),

    topic_key           VARCHAR(50) NOT NULL,                   -- LearningStage 枚举 key
    difficulty          VARCHAR(10) NOT NULL CHECK (difficulty IN ('easy','medium','hard')),

    -- 题干
    label               VARCHAR(255) NOT NULL,                   -- 如 英语·动物词汇·⭐⭐
    question            TEXT NOT NULL,                          -- 题干正文

    -- 选项（JSONB）
    options             JSONB NOT NULL,
    -- [{"id": "a", "text": "Shark", "emoji": "🦈"}, ...]

    -- 答案
    correct_answer_id    VARCHAR(10) NOT NULL,

    -- 解析
    explanation         TEXT NOT NULL,
    explanation_steps   TEXT[] NOT NULL,

    -- 鼓励语
    encouragement       TEXT NOT NULL,

    -- 预估用时 & 奖励
    estimated_minutes  INT NOT NULL DEFAULT 3,
    reward              JSONB NOT NULL DEFAULT '{}',
    -- {"stars": 8, "xp": 18, "xpNeeded": 100, "level": 4, "badge": null}

    -- 成功类型
    success_result_mode VARCHAR(20) CHECK (success_result_mode IN ('correct', 'reward')),

    -- 阶段桥梁标记
    stage_bridge        JSONB,
    -- {"from": "EN_VOCAB_ANIMALS", "to": "EN_VOCAB_FRUITS", "bridgeRatio": 0.4}

    -- LLM 生成元数据
    generation_model    VARCHAR(50),
    generation_prompt   TEXT,
    llm_response_id    VARCHAR(100),

    -- 质量审核
    audit_status        VARCHAR(20) NOT NULL DEFAULT 'pending'
                            CHECK (audit_status IN ('pending','approved','rejected','flagged')),
    audit_note          TEXT,
    audited_at          TIMESTAMPTZ,
    audited_by          UUID,

    -- 正确率统计（定时更新）
    times_shown         INT NOT NULL DEFAULT 0,
    times_correct       INT NOT NULL DEFAULT 0,
    accuracy            DECIMAL(5,2) GENERATED ALWAYS AS (
        CASE WHEN times_shown > 0
             THEN ROUND(times_correct * 100.0 / times_shown, 2)
             ELSE 0 END
    ) STORED,

    tags                TEXT[] NOT NULL DEFAULT '{}',

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
```

// ========== 答题核心逻辑：combo 与 reward 触发机制 ==========
// 位置：content/service/QuizService.java
// 设计要点：
//   1. combo 在 learning_sessions 中追踪（current_combo / longest_combo）
//   2. resultMode 由题目的 success_result_mode + combo 综合决定
//   3. 答对 → current_combo++，答错 → current_combo = 0
//   4. success_result_mode 字段在题目生成时由 LLM 设置（或由后端规则赋值）

/**
 * 题目 success_result_mode 赋值规则
 *
 * 由 QuestionContextBuilder 在调用 LLM 生成题目的同时决定，规则如下：
 *
 * | 条件                          | success_result_mode |
 * |-------------------------------|---------------------|
 * | 随机 ~20% 题目（普通奖励）       | 'reward'            |
 * | 其余 ~80% 题目（标准奖励）       | 'correct'           |
 *
 * - 'reward' 题目奖励更高：stars=10, xp=20, 可能附带徽章
 * - 'correct' 题目奖励标准：stars=8, xp=18, 无徽章
 * - 此比例可在 AppProperties.comboReward 中配置
 */

/**
 * 答题提交 resultMode 计算规则
 *
 * 后端 QuizService.submit() 中：
 *
 * 答错：
 *   resultMode = 'wrong'
 *   current_combo = 0
 *
 * 答对 + success_result_mode == 'reward'：
 *   resultMode = 'reward'   ← 触发连击奖励弹窗
 *   current_combo++
 *   longest_combo = max(longest_combo, current_combo)
 *
 * 答对 + success_result_mode == 'correct'：
 *   resultMode = 'correct'
 *   current_combo++
 *   longest_combo = max(longest_combo, current_combo)
 */

/**
 * 答题奖励配置（详细定义见 8.2.1 AppProperties.java）
 */
@ConfigurationProperties(prefix = "app.quiz")
public record QuizProperties(
    double rewardQuestionRatio,   // reward 题目比例（默认 0.2）
    int baseStars,                // 普通题星星（默认 8）
    int baseXp,                   // 普通题 XP（默认 18）
    int rewardStars,              // 奖励题星星（默认 10）
    int rewardXp,                 // 奖励题 XP（默认 20）
    int wrongDeductionStars,       // 答错扣星星（默认 0）
    int dailyStarsLimit           // 每日星星上限（默认 50）
) {
    public QuizProperties {
        if (rewardQuestionRatio <= 0 || rewardQuestionRatio > 1) rewardQuestionRatio = 0.2;
        if (baseStars <= 0) baseStars = 8;
        if (baseXp <= 0) baseXp = 18;
        if (rewardStars <= 0) rewardStars = 10;
        if (rewardXp <= 0) rewardXp = 20;
        if (dailyStarsLimit <= 0) dailyStarsLimit = 50;
    }
}

/**
 * QuizService.doSubmit 伪代码
 *
 * ```java
 * private QuizSubmitResponse doSubmit(QuizSubmitRequest req) {
 *     // 1. 查询题目
 *     Question q = questionRepository.findById(req.questionId())
 *         .orElseThrow(() -> new EntityNotFoundException("Question", req.questionId()));
 *
 *     // 2. 判定对错
 *     boolean isCorrect = q.getCorrectAnswerId().equals(req.answerId());
 *
 *     // 3. 获取当前会话（根据 childId 查询或创建）
 *     LearningSession session = getOrCreateSession(req, childId);
 *
 *     // 4. 更新 combo 计数器
 *     if (isCorrect) {
 *         session.setCurrentCombo(session.getCurrentCombo() + 1);
 *         session.setLongestCombo(max(session.getLongestCombo(), session.getCurrentCombo()));
 *     } else {
 *         session.setCurrentCombo(0);  // 答错归零
 *     }
 *
 *     // 5. 确定 resultMode（答题对错 + 题目类型）
 *     String resultMode = isCorrect
 *         ? q.getSuccessResultMode()   // 'reward' 或 'correct'
 *         : "wrong";
 *
 *     // 6. 计算奖励（由后端根据 resultMode 决定 stars/xp/是否发徽章）
 *     RewardDTO reward = computeReward(q, session, resultMode);
 *
 *     // 7. 写入 quiz_answers（含 combo 快照）
 *     QuizAnswer answer = QuizAnswer.builder()
 *         .sessionId(session.getId())
 *         .questionId(q.getId())
 *         .planQuestionId(req.planQuestionId())   // 可选
 *         .selectedAnswer(req.answerId())
 *         .isCorrect(isCorrect)
 *         .timeSpentSec(req.timeSpent())
 *         .comboAtAnswer(session.getCurrentCombo())  // ← 快照：答题时的 combo
 *         .snapshotTopic(q.getTopic())
 *         .snapshotStageKey(q.getTopicKey())
 *         .answeredAt(now())
 *         .build();
 *     quizAnswerRepository.save(answer);
 *
 *     // 8. 更新会话统计
 *     session.setTotalQuestions(session.getTotalQuestions() + 1);
 *     if (isCorrect) session.setCorrectCount(session.getCorrectCount() + 1);
 *     session.setXpEarned(session.getXpEarned() + reward.xp());
 *     session.setStarsEarned(session.getStarsEarned() + reward.stars());
 *     learningSessionRepository.save(session);
 *
 *     // 9. 更新孩子表累计星星
 *     childRepository.addStars(childId, reward.stars());
 *     childRepository.addXp(childId, reward.xp());
 *
 *     // 10. 触发成就判定（发送 ApplicationEvent）
 *     applicationEventPublisher.publishEvent(new AnswerSubmittedEvent(childId, session, q, isCorrect));
 *
 *     // 11. 判断是否还有下一题
 *     boolean hasMore = questionRepository.existsUnanswered(childId, session.getPlanId());
 *
 *     // 12. 返回结果
 *     return new QuizSubmitResponse(
 *         isCorrect,
 *         resultMode,
 *         q.getCorrectAnswerId(),
 *         q.getExplanation(),
 *         List.copyOf(q.getExplanationSteps()),
 *         q.getEncouragement(),
 *         !isCorrect || "reward".equals(resultMode) || !hasMore,  // showResult
 *         hasMore,
 *         new ResultDTO(
 *             session.getCurrentCombo(),
 *             calcAccuracy(session),
 *             req.timeSpent(),
 *             reward.stars(),
 *             todayMaxStars(childId),
 *             reward.xp(),
 *             reward.xpNeeded(),
 *             reward.level(),
 *             reward.badge()
 *         )
 *     );
 * }
 * ```
 */

```java
// com.kidslearn.content.service.QuestionContextBuilder
@Service
public class QuestionContextBuilder {

    public record QuestionGenerationContext(
        // 孩子基础信息
        UUID childId,
        String childName,
        int age,
        String ageGroup,
        // 当前阶段
        LearningStage currentStage,
        LearningStage nextStage,
        // 历史表现
        int totalAnswered,
        int totalCorrect,
        BigDecimal accuracyRate,
        int currentStreak,
        // 今日答题
        int todayAnswered,
        int todayCorrect,
        // 当前会话
        int sessionCombo,
        int sessionCorrect,
        // 目标
        String topic,         // 本次要考的主题
        String difficulty,    // easy / medium / hard
        int targetAccuracy   // 期望正确率 60-95
    ) {}

    public QuestionGenerationContext buildContext(
        Child child,
        Question lastQuestion,
        QuizSession session,
        String targetTopic,
        String difficulty
    ) {
        // 计算当前阶段
        LearningStage currentStage = determineCurrentStage(child, session);
        LearningStage nextStage = determineNextStage(child, currentStage);

        // 计算今日正确率
        int todayAnswered = session.getTodayAnswered();
        int todayCorrect = session.getTodayCorrect();
        int targetAccuracy = calculateTargetAccuracy(
            child, currentStage, session.getCombo()
        );

        return new QuestionGenerationContext(
            child.getId(),
            child.getName(),
            child.getAge(),
            child.getAgeGroup(),
            currentStage,
            nextStage,
            child.getTotalAnswered(),
            child.getTotalCorrect(),
            child.getAccuracyRate(),
            child.getStreak(),
            todayAnswered,
            todayCorrect,
            session.getCombo(),
            session.getCorrectCount(),
            targetTopic,
            difficulty,
            targetAccuracy
        );
    }

    /** 根据孩子历史正确率和连击数，动态调整目标正确率 */
    private int calculateTargetAccuracy(Child child, LearningStage stage, int combo) {
        BigDecimal base = child.getAccuracyRate();

        // 连击加成：连击越高，目标正确率适当降低（题目难度提升）
        int comboBonus = Math.min(combo * 2, 15); // 最多+15%

        // 阶段加成：越高级别目标越高
        int stageBonus = stage.ordinal() * 3; // 每个阶段+3%

        int target = base.intValue() + comboBonus + stageBonus;
        return Math.min(95, Math.max(55, target)); // 55-95 范围
    }
}
```

```markdown
## 角色定义
你是一个专为 3-14 岁儿童设计学习题目的 AI 出题助手。
你的职责是根据孩子当前所处学习阶段和下一阶段目标，

### 1. 阶段递进原则
每个题目必须明确其在**阶段桥梁**中的位置：
- **巩固题 (30%)**：直接考察当前阶段核心知识点
- **桥梁题 (40%)**：同时包含当前阶段和下一阶段元素，自然过渡
- **挑战题 (30%)**：略微超出当前阶段，启发思维

### 2. 适龄原则
题目设计必须匹配孩子年龄：
| 年龄段 | 认知特点 | 设计要点 |
|--------|---------|---------|
| 3-5岁 | 具体形象思维 | 用图片emoji辅助，选项简短 |
| 6-8岁 | 具象运算思维 | 贴近生活场景，步骤分解 |

- 严禁：暴力、血腥、性暗示、政治、歧视内容
- 严禁：真实人名/地名/学校名
- 所有场景：学校、家庭、动物园、游乐场、外太空、海底世界等儿童友好场景

| 难度 | 目标正确率 | 特点 |
|------|-----------|------|
| easy | 85-95% | 单一知识点，干扰项明显 |
| medium | 70-85% | 2-3个知识点混合，需简单推理 |
| hard | 55-70% | 多步骤推理，干扰项具有迷惑性 |

1. 严格输出 JSON 格式，不要有额外文本
2. explanationSteps 必须包含 2-4 步
3. encouragement 必须点名孩子的进步点
4. 选项文字要简洁（不超过20字）
```

```java
// com.kidslearn.llm.skills.SkillsLoader
@Component
public class SkillsLoader {

    private static final Path SKILLS_DIR =
        Path.of("src/main/resources/skills");

    private final Map<String, SkillDefinition> skills = new LinkedHashMap<>();

    @PostConstruct
    public void loadAll() throws IOException {
        // 加载注册清单
        ObjectMapper yaml = new ObjectMapper(new YAMLFactory());
        Path registryPath = SKILLS_DIR.resolve("skill.yaml");
        SkillRegistry registry = yaml.readValue(registryPath.toFile(), SkillRegistry.class);

        // 按 priority 排序加载
        registry.getRegistry().stream()
            .filter(SkillDef::isEnabled)
            .sorted(Comparator.comparingInt(SkillDef::getPriority))
            .forEach(this::loadSkill);
    }

    private void loadSkill(SkillDef def) {
        try {
            Path skillDir = SKILLS_DIR.resolve(def.getSlug());
            SkillDefinition skill = SkillDefinition.builder()
                .slug(def.getSlug())
                .name(def.getName())
                .description(def.getDescription())
                .version(def.getVersion())
                .systemPrompt(loadText(skillDir.resolve("prompt-system.md")))
                .userPromptTemplate(loadText(skillDir.resolve("prompt-user.md")))
                .inputSchema(loadJson(skillDir.resolve("skill.yaml"), "inputSchema"))
                .outputSchema(loadJson(skillDir.resolve("skill.yaml"), "outputSchema"))
                .examples(loadExamples(skillDir.resolve("examples")))
                .build();

            skills.put(def.getSlug(), skill);
        } catch (Exception e) {
            log.warn("Failed to load skill {}: {}", def.getSlug(), e.getMessage());
        }
    }

    public SkillDefinition get(String slug) {
        return skills.get(slug);
    }

    /** 按 priority 顺序获取所有已启用的 Skills */
    public List<SkillDefinition> allEnabled() {
        return skills.values().stream().toList();
    }
}
```

```java
            .put("today_accuracy", today != null ? today.getAccuracy().intValue() : 0)
```

```java
            // 获取当前阶段的下一可选阶段

                .orElseThrow(() -> new ResourceNotFoundException("Stage", currentStageKey));
            stages = stageRepository.findByIdIn(current.getNextStageKeys());
        } else if (subject != null) {
            // 按学科筛选适合年龄的阶段
            stages = stageRepository.findBySubjectAndAgeRange(subject, age);
        } else {
            stages = stageRepository.findAll();
        }

        ArrayNode result = objectMapper.createArrayNode();
        for (var stage : stages) {
            result.add(objectMapper.createObjectNode()
                .put("key", stage.getId())
                .put("subject", stage.getSubject())
                .put("topic", stage.getTopic())
                .put("minAge", stage.getMinAge())
                .put("maxAge", stage.getMaxAge())
                .putPOJO("nextStages", stage.getNextStageKeys())
            );
```

### 7.4 LLM 客户端（Claude API）

```java
// com.kidslearn.llm.LlmClient
@Service
public class LlmClient {

    private final AppProperties props;
    private final RestClient restClient;
    private final ToolRegistry toolRegistry;
    private final ObjectMapper objectMapper;

    public PlanGenerateResult generatePlan(String prompt, ToolContext ctx) {
        List<ToolChoice> tools = buildToolChoices();

        LlmRequest request = LlmRequest.of(
            SystemPrompt.PLAN_GENERATE,
            prompt,
            tools,
            ctx
        );

        LlmResponse resp = callApi(request);
        return parsePlanResult(resp);
    }

    private List<ToolChoice> buildToolChoices() {
        return toolRegistry.all().stream()
            .map(tool -> new ToolChoice(
                tool.name(),
                tool.description(),
                tool.inputSchema()
            ))
            .toList();
    }

    private LlmResponse callApi(LlmRequest req) {
        // 使用 RestClient 调用 Claude Messages API
        return restClient.post()
            .uri("/v1/messages")
            .header("x-api-key", props.getLlm().getApiKey())
            .header("anthropic-version", "2023-06-01")
            .contentType(APPLICATION_JSON)
            .body(req)
            .retrieve()
            .body(LlmResponse.class);
    }
}

public record LlmRequest(
    String model,
    List<Message> messages,
    int maxTokens,
    List<ToolChoice> tools
) {
    public record Message(String role, String content) {}
    public record ToolChoice(String name, String description, JsonNode inputSchema) {}

    public static LlmRequest of(String systemPrompt, String userPrompt,
                                 List<ToolChoice> tools, ToolContext ctx) {
        return new LlmRequest(
            "claude-sonnet-4-7-20260219",
            List.of(
                new Message("system", systemPrompt),
                new Message("user", buildContextPrompt(ctx) + "\n\n" + userPrompt)
            ),
            4096,
            tools
        );
    }

    private static String buildContextPrompt(ToolContext ctx) {
        return """
            当前上下文:
            - 家长ID: %s
            - 孩子ID: %s
            - 孩子姓名: %s
            - 孩子年龄: %d 岁
            - 年龄段: %s
            """.formatted(
                ctx.parentId(), ctx.childId(), ctx.childName(),
                ctx.childAge(), ctx.childAgeGroup()
            );
    }
}
```

```java
class VirtualThreadLoadTest {

    @Test
    @DisplayName("100并发虚拟线程答题压测")
    void submitAnswer_concurrent100_success() {
        int concurrency = 100;
        int requestsPerThread = 10;
        CountDownLatch latch = new CountDownLatch(concurrency);

        ExecutorService executor = Executors.newVirtualThreadPerTaskExecutor();

        long start = System.currentTimeMillis();
        AtomicInteger successCount = new AtomicInteger(0);
        AtomicInteger errorCount = new AtomicInteger(0);

        for (int t = 0; t < concurrency; t++) {
            executor.submit(() -> {
                try {
                    for (int i = 0; i < requestsPerThread; i++) {
                        try {
                            // 模拟答题请求
                            quizService.submitAnswer(
                                new QuizSubmitRequest(questionId, null, answerId, 30),
                                childId
                            );
                            successCount.incrementAndGet();
                        } catch (Exception e) {
                            errorCount.incrementAndGet();
                        }
                    }
                } finally {
                    latch.countDown();
                }
            });
        }

        latch.await(60, TimeUnit.SECONDS);
        long duration = System.currentTimeMillis() - start;

        System.out.printf("""
            Virtual Thread Load Test Results:
            - Concurrency: %d
            - Total Requests: %d
            - Success: %d
            - Errors: %d
            - Duration: %d ms
            - Throughput: %.2f req/s
            """,
            concurrency,
            concurrency * requestsPerThread,
            successCount.get(),
            errorCount.get(),
            duration,
            (concurrency * requestsPerThread) * 1000.0 / duration
        );

        assertThat(errorCount.get()).isLessThan(concurrency * requestsPerThread * 0.01); // <1% error
    }
}
```
